# =============================================================
# DATABASE SETUP (PostgreSQL)
# =============================================================
# Run these SQL files in order against your 'railpulse' database:
#
#   1. database/schema.sql              -- Creates all tables
#   2. database/add_stations_trains.sql -- Seeds stations & train_details
#   3. database/add_train_stops.sql     -- Seeds train stop schedules
#   4. database/add_username.sql        -- Adds username column to users (migration)
#
# Commands:
#   psql -U postgres -d railpulse -f database/schema.sql
#   psql -U postgres -d railpulse -f database/add_stations_trains.sql
#   psql -U postgres -d railpulse -f database/add_train_stops.sql
#   psql -U postgres -d railpulse -f database/add_username.sql
#
# NOTE: If setting up fresh, the schema.sql already includes the username column.
#       The add_username.sql migration is only needed for existing databases.


# =============================================================
# USER ROLES SUPPORTED
# =============================================================
#
#   Passenger         -- Files complaints, views own complaints, crowd monitoring
#   Admin             -- Full access: all complaints, system overview
#   Station Master    -- Station-type complaints only
#   TT                -- Train-type complaints only (Ticket Examiner)
#   Loco Pilot        -- Train-type complaints only
#   Guard             -- Train-type complaints only
#   RPF Officer       -- All complaint types (Railway Protection Force)
#   Section Engineer  -- Station-type complaints only
#
# Login Credential: username (unique User ID set at registration) + password


# =============================================================
# ENVIRONMENT VARIABLES REQUIRED
# =============================================================
# 1. Copy .env.example to .env
# 2. Update DATABASE_URL with your local PostgreSQL credentials:
#
#    DATABASE_URL=postgresql://<user>:<password>@localhost:5432/railpulse
#    JWT_SECRET=any_strong_random_string
#
# The backend reads this file via dotenv.
# Frontend API calls default to http://localhost:5000
